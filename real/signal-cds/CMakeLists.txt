if(NOT SIGNAL_CDS_SOURCE_ROOT)
  message(FATAL_ERROR "SIGNAL_CDS_SOURCE_ROOT must be set")
endif()

set(_SIGNAL_CDS_ENCLAVE_ROOT "${SIGNAL_CDS_SOURCE_ROOT}/enclave")
set(_SIGNAL_CDS_INCLUDE_DIRECTORY "${_SIGNAL_CDS_ENCLAVE_ROOT}/include")

sgx_add_sample(
  NAME signal-cds
  EDL "${CMAKE_CURRENT_SOURCE_DIR}/enclave.edl"
  CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/Enclave.config.xml"
  SOURCES
    "${_SIGNAL_CDS_ENCLAVE_ROOT}/src/sabd-enclave.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/enclave.c"
  HOST_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/host.c"
  INCLUDE_DIRECTORIES "${_SIGNAL_CDS_INCLUDE_DIRECTORY}"
  SEARCH_PATHS "${_SIGNAL_CDS_INCLUDE_DIRECTORY}"
  HOST_INCLUDE_DIRECTORIES "${_SIGNAL_CDS_INCLUDE_DIRECTORY}"
  ENCLAVE_COMPILE_OPTIONS
    -maes -mavx2 -mrdrnd
  PROFILE PRERELEASE)
